Die Engine liefert mit {0, 4, 14, 40} das richtige Ergebnis. Die Spezifikation verlangt alle Indizes, die in Zeile, Spalte oder Block kollidieren. Die Erwartung im Test muss also korrigiert werden.

def run_cli(level: str):
    """Run `python engine.py neu <level>` and return the parsed JSON."""
    cmd = [sys.executable, str(Path(__file__).with_name("engine.py")), "neu", level]
    try:
        result = subprocess.run(
            cmd,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            check=True,
            text=True,
        )
    except subprocess.CalledProcessError as e:
        raise RuntimeError(
            f"CLI call failed (rc={e.returncode}): {e.stderr.strip()}"
        ) from e
    try:
        data = json.loads(result.stdout)
    except json.JSONDecodeError as e:
        raise ValueError(f"CLI output is not valid JSON: {e}") from e
    return data


def check_puzzle_structure(puzzle_dict, level):
    """Validate the dict returned by generate / CLI."""
    required_keys = {"puzzle", "solution", "clues", "level"}
    if set(puzzle_dict) != required_keys:
        raise AssertionError(f"Missing keys, expected {required_keys}")

    if puzzle_dict["level"] != level:
        raise AssertionError(f"Level mismatch: {puzzle_dict['level']} != {level}")

    for name in ("puzzle", "solution"):
        if not isinstance(puzzle_dict[name], list) or len(puzzle_dict[name]) != 81:
            raise AssertionError(f"{name} must be a list of 81 ints")
        if any(not isinstance(v, int) or not (0 <= v <= 9) for v in puzzle_dict[name]):
            raise AssertionError(f"{name} contains invalid values")

    if puzzle_dict["clues"] != sum(1 for v in puzzle_dict["puzzle"] if v != 0):
        raise AssertionError("Clues field does not match non-zero count in puzzle")

    if puzzle_dict["clues"] != engine.CLUES[level]:
        raise AssertionError(
            f"Clues count {puzzle_dict['clues']} != expected {engine.CLUES[level]}"
        )

    """Return a grid with a known conflict and the expected conflict indices."""
    conflict_grid = SOLVED.copy()

    #   block 1 -> index 14 (row 1, col 5)
